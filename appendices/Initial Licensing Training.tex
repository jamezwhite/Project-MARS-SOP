\chapter{Initial Licensing Training}
\label{ch:ilt}

    \claudeadd{This appendix establishes the grading rubric and evaluation standards for Initial Licensing Training, the course of instruction required for graduation from the \projectmars{} \ac{suas} Operator Course referenced in \nameref{sec:qual-qp}. Successful completion of Initial Licensing Training, culminating in a passing score on the Licensing Evaluation Flight described in \nameref{sec:ilt-evaluation}, is a prerequisite for award of \ac{qp} qualification.}

    \claudeadd{The operator's manual remains the governing authority for operation of the aircraft. Where the task standards in this appendix conflict with the operator's manual, this appendix is the governing authority for training and evaluation purposes only, consistent with \nameref{ch:certification}.}

\section{Training Pathway}\label{sec:ilt-pathway}

    \claudeadd{Initial Licensing Training progresses candidates through three sequential phases. A candidate must successfully complete each phase before advancing to the next.}

    \begin{enumerate}
        \item \claudeadd{\textbf{Phase I --- Academics and Written Examination} (\nameref{sec:ilt-academics}): Classroom instruction, a written examination, and a hands-on systems proficiency demonstration.}
        \item \claudeadd{\textbf{Phase II --- Simulated and Supervised Practice} (\nameref{sec:ilt-practice}): Simulated and supervised live practice of all Initial Licensing Training tasks under instructor supervision.}
        \item \claudeadd{\textbf{Phase III --- Licensing Evaluation Flight} (\nameref{sec:ilt-evaluation}): A formal, graded evaluation flight culminating in award of the \projectmars{} \ac{suas} Operator License.}
    \end{enumerate}

\section{Phase I: Academics and Written Examination}\label{sec:ilt-academics}

    \subsection{Academic Instruction}\label{sec:ilt-academics-instruction}

        \claudeadd{Candidates will receive resident academic instruction on, at minimum, the following topics prior to the written examination:}

        \begin{itemize}
            \item \claudeadd{Regulatory and policy (AR 95-1, this \ac{sop}, and applicable local directives)}
            \item \claudeadd{Airspace}
            \item \claudeadd{Safety and \ac{crm}}
            \item \claudeadd{Weather and its effects on flight}
            \item \claudeadd{Aerodynamics}
            \item \claudeadd{Mission planning}
            \item \claudeadd{Communications and \ac{gcs} data links}
            \item \claudeadd{Aircraft systems and emergency procedures}
            \item \claudeadd{Record keeping}
        \end{itemize}

    \subsection{Written Examination Standard}\label{sec:ilt-writtenexam}

        \claudeadd{The written examination will consist of a minimum of 50 questions, with at least 10 questions covering each topic listed in \nameref{sec:ilt-academics-instruction}. A candidate must score at least 70 percent on every individual topic to pass. A candidate who does not achieve 70 percent on a topic will retest only the topic(s) failed.}

    \subsection{Hands-On Systems Proficiency}\label{sec:ilt-handson}

        \claudeadd{In addition to the written examination, candidates will demonstrate 100 percent proficiency in the following hands-on tasks before advancing to Phase II:}

        \begin{itemize}
            \item \claudeadd{Identification of \projectmars{} airframe components and characteristics}
            \item \claudeadd{Assembly, battery installation, and \ac{gcs} pairing in accordance with \nameref{cl:preflight}}
            \item \claudeadd{\ac{medlog} payload handling and securing procedures}
            \item \claudeadd{Operator-level pre-flight and post-flight inspection}
        \end{itemize}

\section{Phase II: Simulated and Supervised Practice}\label{sec:ilt-practice}

    \claudeadd{Following successful completion of Phase I, candidates will complete simulated and supervised live practice of Tasks IL-1 through IL-11 (\nameref{sec:ilt-tasks}) under the direct supervision of a \ac{qp}- or \ac{eo}-qualified instructor, using a compatible simulator and the assigned \projectmars{} airframe, until the instructor assesses the candidate as ready to attempt the Licensing Evaluation Flight.}

    \claudeadd{Phase II practice is developmental and is not independently graded. Readiness to advance to Phase III is determined by the supervising instructor's assessment.}

\section{Phase III: Licensing Evaluation Flight}\label{sec:ilt-evaluation}

    \subsection{Scoring Standard}\label{sec:ilt-evaluation-scoring}

        \claudeadd{The Licensing Evaluation Flight consists of Tasks IL-1 through IL-10 (\nameref{sec:ilt-tasks}), plus the mandatory Task IL-11, Emergency Procedures and Manual Override.}

        \claudeadd{Each of Tasks IL-1 through IL-10 is graded GO or NO-GO against every performance measure listed for that task. A candidate who achieves a GO on 100 percent of a task's performance measures receives a GO and 100 points for that task; a NO-GO on any single performance measure results in a NO-GO and zero points for that task. The maximum possible score across Tasks IL-1 through IL-10 is 1000 points.}

        \claudeadd{To pass the Licensing Evaluation Flight and be awarded the \projectmars{} \ac{suas} Operator License, a candidate must ---}

        \begin{itemize}
            \item \claudeadd{Achieve a GO on Task IL-11, Emergency Procedures and Manual Override. A NO-GO on Task IL-11 is an automatic failure of the Licensing Evaluation Flight, regardless of the candidate's performance on Tasks IL-1 through IL-10.}
            \item \claudeadd{Achieve a GO on at least 7 of the 10 tasks (Tasks IL-1 through IL-10).}
            \item \claudeadd{Achieve a cumulative score of at least 700 of the 1000 points possible across Tasks IL-1 through IL-10.}
        \end{itemize}

    \subsection{Retraining and Re-Evaluation}\label{sec:ilt-evaluation-retrain}

        \claudeadd{A candidate who does not meet the standard in \nameref{sec:ilt-evaluation-scoring} will retrain and re-fly only the task(s) on which a NO-GO was received; tasks already scored GO are not re-flown.}

        \claudeadd{A candidate's rating upon successful completion is recorded as ``Licensed'' if every task was scored GO on the first attempt, or ``Licensed, Q\{n\}'' where \{n\} is the highest number of attempts required to achieve a GO on any single task. For example, a candidate who requires three attempts to achieve a GO on Task IL-7 is rated ``Licensed, Q3.'' No candidate is rated higher than ``Licensed,'' regardless of performance on the first attempt.}

    \subsection{Evaluators}\label{sec:ilt-evaluation-evaluators}

        \claudeadd{Licensing Evaluation Flights will be conducted by an \ac{eo}-qualified evaluator in accordance with \nameref{sec:qual-eo}. Where more than one \ac{eo}-qualified evaluator is available, the evaluating \ac{eo} will not be the same \ac{eo} who provided the candidate's Phase II supervised practice instruction, to preserve the objectivity of the evaluation.}

\section{Task Standards}\label{sec:ilt-tasks}

    \subsection{General Task Standard}\label{sec:ilt-tasks-general}

        \claudeadd{Tasks IL-1 through IL-10 share the following task, condition, and standard, and are evaluated according to the performance measures listed for each task below.}

        \begin{itemize}
            \item \claudeadd{\textbf{Task:} Conduct a simulated or live flight in accordance with the performance measures for the task.}
            \item \claudeadd{\textbf{Condition:} As a \projectmars{} \ac{suas} Operator Course candidate in a classroom or operational environment, with a \ac{gcs}, \projectmars{} airframe, applicable operator's manual, this \ac{sop}, and local range procedures.}
            \item \claudeadd{\textbf{Standard:} Conduct a simulated or live flight in accordance with the task's performance measures, without safety violations and with a GO on 100 percent of the performance measures listed.}
        \end{itemize}

    \subsection{Task IL-1 --- Takeoff}\label{task:il-1}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Increase throttle to provide sufficient power for the aircraft to lift off the ground.}
            \item \claudeadd{Ascend vertically in a controlled manner to a safe hover altitude (10--20 feet \ac{agl}).}
            \item \claudeadd{Establish a stable hover before transitioning to forward flight.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-2 --- Hover}\label{task:il-2}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Maintain position over a designated ground reference point with minimal drift, accounting for wind.}
            \item \claudeadd{Make smooth, minimal control inputs to maintain stability.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-3 --- Accelerate}\label{task:il-3}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Establish a hover.}
            \item \claudeadd{Adjust pitch and throttle setting to enter forward flight.}
            \item \claudeadd{Maintain altitude throughout the maneuver.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-4 --- Level Flight}\label{task:il-4}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Establish and maintain an assigned altitude while maintaining forward flight.}
            \item \claudeadd{Establish and maintain an assigned heading.}
            \item \claudeadd{Make smooth and timely corrections to maintain the intended direction of flight or flight path.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-5 --- Employ System Capabilities}\label{task:il-5}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Continuously monitor battery voltage/remaining capacity, \ac{gps} status, and data link status displayed in the \ac{gcs} software.}
            \item \claudeadd{Verify the aircraft remains within the programmed \term{geofence} boundary throughout flight.}
            \item \claudeadd{Verify the \ac{medlog} payload remains secure in the payload compartment throughout flight.}
            \item \claudeadd{Identify and correctly respond to any low-battery, geofence, or link-loss indication in accordance with this \ac{sop}.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-6 --- Decelerate}\label{task:il-6}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Maintain straight and level flight.}
            \item \claudeadd{Adjust pitch and throttle setting to reduce speed.}
            \item \claudeadd{Establish a hover.}
            \item \claudeadd{Maintain altitude throughout the maneuver.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-7 --- Land}\label{task:il-7}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Select the designated landing point.}
            \item \claudeadd{Reduce speed while orienting on the designated landing point.}
            \item \claudeadd{Enter a hover.}
            \item \claudeadd{Using smooth control inputs, descend toward the intended landing point.}
            \item \claudeadd{Land on the intended landing point.}
            \item \claudeadd{Confirm the catcher is positioned at the landing site and has acknowledged receipt of the \ac{medlog} payload prior to shutdown, in accordance with \nameref{cl:preflight}.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}

    \subsection{Task IL-8 --- Box Drills}\label{task:il-8}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Maintain straight and level flight.}
            \item \claudeadd{Begin a turn by inputting roll left or right in the direction of choice.}
            \item \claudeadd{Apply yaw in the direction of turn as needed to maintain the nose of the aircraft in the direction of turn.}
            \item \claudeadd{Enter straight and level flight after completion of the turn.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
            \item \claudeadd{Complete two iterations of each direction of travel.}
        \end{itemize}

    \subsection{Task IL-9 --- Figure 8}\label{task:il-9}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Maintain straight and level flight, maneuvering toward the center point of the figure-8.}
            \item \claudeadd{Cross from left to right, or vice versa, at the center point.}
            \item \claudeadd{Apply yaw in the direction of turn to orbit around the corner point.}
            \item \claudeadd{Perform the same maneuver in the opposite direction.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
            \item \claudeadd{Complete two full figure-8 drills.}
        \end{itemize}

    \subsection{Task IL-10 --- Orbit}\label{task:il-10}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Establish a target point to orbit.}
            \item \claudeadd{Orient the aircraft on the target.}
            \item \claudeadd{Apply yaw in the direction of orbit as needed to maintain the nose of the aircraft, or sensor, on the target.}
            \item \claudeadd{Apply roll in the opposite direction of orbit as needed to maintain direction of flight around the target.}
            \item \claudeadd{Maintain altitude throughout the maneuver.}
            \item \claudeadd{Enter straight and level flight after completion of the orbit.}
            \item \claudeadd{Avoid all obstacles.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
            \item \claudeadd{Complete two full orbits.}
        \end{itemize}

    \subsection{Task IL-11 --- Emergency Procedures and Manual Override}\label{task:il-11}

        \claudeadd{Task IL-11 does not share the general task standard in \nameref{sec:ilt-tasks-general} and is instead evaluated as follows.}

        \begin{itemize}
            \item \claudeadd{\textbf{Task:} Recognize and respond to a simulated emergency or off-nominal condition presented by the evaluator during the Licensing Evaluation Flight.}
            \item \claudeadd{\textbf{Condition:} During a live evaluation flight, presented with a simulated emergency or off-nominal condition (for example, simulated link loss, low battery, geofence breach, or evaluator direction to take manual control) by the evaluator.}
            \item \claudeadd{\textbf{Standard:} Correctly recognize and respond to the simulated emergency in accordance with \nameref{cl:emergencyprocedures} and this \ac{sop}, with a GO on 100 percent of the performance measures below. A NO-GO on this task is an automatic failure of the Licensing Evaluation Flight regardless of the candidate's cumulative score on Tasks IL-1 through IL-10, in accordance with \nameref{sec:ilt-evaluation-scoring}.}
        \end{itemize}

        \claudeadd{Performance measures:}

        \begin{itemize}
            \item \claudeadd{Correctly identify the simulated emergency or off-nominal condition.}
            \item \claudeadd{Take direct manual control of the aircraft without undue delay, if required by the emergency.}
            \item \claudeadd{Execute the appropriate response in accordance with \nameref{cl:emergencyprocedures}.}
            \item \claudeadd{Communicate the emergency and the response taken to the \ac{gcs} operator and ground personnel.}
            \item \claudeadd{Return the aircraft to a safe, stable flight condition, or execute a safe landing, as appropriate.}
            \item \claudeadd{Maintain situational awareness of aircraft operation to ensure safety at all times.}
        \end{itemize}
